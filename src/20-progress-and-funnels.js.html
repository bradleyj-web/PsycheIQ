/* ============================================================ PSYCHEIQ — PROGRESS SPINE + MILESTONE FUNNELS ------------------------------------------------------------ One progress system, not five. The app already computed an "experience level" in growthStanding() and then barely showed it; this makes that visible and gives it a continuous bar. The chain is deliberately linear: XP -> level -> sigils -> deck skins XP is earned, levels are derived from XP, levelling pays out sigils, and sigils are the only spendable currency (they already buy deck skins in module 18). Badges stay as rare discrete markers. Nothing else keeps score. Growth *stage* (growthStanding().stage) is a different axis — it is derived from the composite score, not from activity — so it stays as flavour text and is not folded in here. NOTE ON GATING: nothing a visitor might have arrived for is locked behind a level. Levels unlock depth and cosmetics, never doors. Someone who came from social media for a tarot pull gets their tarot pull. ============================================================ */ /* what earns XP — all of these are things the app already tracks */ const XP_RULES = { assessment: 100, // completing any of the eight chapter: 50, // a Fool's Journey chapter trial: 20, // the daily challenge reading: 10, // a deck reading streak: 5 // each day a streak is held }; /* Level curve: each level costs a little more than the last, so the bar always moves early and never stalls completely later. Base is deliberately NOT a multiple of the assessment award — at 100 the first assessment landed exactly on a level boundary and the bar read "0 / 160" right after earning 100 XP, which looks like nothing happened. At 150 the first assessment fills two thirds of the bar. */ const XP_STEP_BASE = 150, XP_STEP_GROWTH = 60; const xpNeededFor = lvl => XP_STEP_BASE + (lvl - 1) * XP_STEP_GROWTH; /* The four named roles from module 11 become bands over the numeric level, so there is still only one ladder. */ function xpTier(level){ const i = level >= 7 ? 3 : level >= 5 ? 2 : level >= 3 ? 1 : 0; return LEVELS[i]; } function xpLevelFor(xp){ let level = 1, rest = Math.max(0, xp | 0); while (rest >= xpNeededFor(level) && level < 99){ rest -= xpNeededFor(level); level++; } const need = xpNeededFor(level); const tier = xpTier(level); return {level, into: rest, need, pct: Math.round(rest / need * 100), name: tier.n, glyph: tier.g, line: tier.l, total: xp | 0}; } /* ---------------- state ---------------- */ const XPState = {xp:0, level:1, seen:1}; // `seen` = highest level already celebrated async function loadXP(){ const x = await Store.get("pq:xp"); if (x){ XPState.xp = x.xp | 0; XPState.seen = x.seen || 1; } syncXPBaseline(); XPState.level = xpLevelFor(XPState.xp).level; paintXP(); } function saveXP(){ return Store.set("pq:xp", {xp:XPState.xp, seen:XPState.seen}); } /* Anyone who already finished assessments before this shipped should not open the app on level 1 with an empty bar. Floor their XP at what those completions are worth, without ever subtracting. */ function syncXPBaseline(){ const done = TESTIDS.filter(id => State[id]).length; const floor = done * XP_RULES.assessment; if (XPState.xp < floor){ XPState.xp = floor; XPState.seen = Math.max(XPState.seen, xpLevelFor(floor).level); // no retroactive level-up spam } } /* ---------------- earning ---------------- */ function awardXP(kind, opts){ const amount = XP_RULES[kind]; if (!amount) return null; const o = opts || {}; const before = xpLevelFor(XPState.xp); XPState.xp += amount; const after = xpLevelFor(XPState.xp); XPState.level = after.level; const levelled = after.level > before.level && after.level > XPState.seen; if (levelled){ XPState.seen = after.level; /* Levelling is the only thing that mints sigils outside the Journey, which is what keeps XP and sigils one chain rather than two competing currencies. Toast instead. */ if (o.quiet) toast("Level " + after.level + " · " + after.name + " · +" + payout + " sigils"); else setTimeout(() => levelUpModal(after, payout), o.delay || 900); } else if (!o.quiet && !o.silentToast){ toast("+" + amount + " XP"); } saveXP(); paintXP(); return {amount, levelled, level: after.level}; } function levelUpModal(lv, payout){ if (typeof burstCentre === "function") burstCentre({count:52, spread:300, life:1500}); modal( '

Mind+Body+Soul -> the Journey is the reason to subscribe Each fires once ever, per profile. progressFunnel() returns true when it has taken the screen, so afterTest() can skip the paywall modal rather than stacking two dialogs on top of each other. */ function funnelState(){ const p = activeP(); if (!p.funnels) p.funnels = {}; return p.funnels; } function funnelFired(k){ return !!funnelState()[k]; } function markFunnel(k){ funnelState()[k] = true; persistProfiles(); } function nextSuggested(){ /* prefer an untouched dimension, so the Blueprint fills out across Mind/Body/Soul rather than three deep in one of them */ const p = dimProgress(); const thin = ["Mind","Body","Soul"] .filter(d => p[d] && p[d].done === 0) .sort((a,b) => (p[b].total - p[a].total)); const pool = TESTS.filter(t => !t.done()); if (!pool.length) return null; if (thin.length){ const hit = pool.find(t => t.group === thin[0]); if (hit) return hit; } return pool[0]; } function progressFunnel(id){ const done = TESTIDS.filter(x => State[x]).length; const p = dimProgress(); const dims = ["Mind","Body","Soul"].filter(d => p[d] && p[d].done > 0).length; /* --- beat 3: a baseline in all three dimensions --- */ if (dims === 3 && !funnelFired("journey")){ markFunnel("journey"); const nxt = chapterOf(0); modal( '
