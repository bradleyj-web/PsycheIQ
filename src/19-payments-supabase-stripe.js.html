window.supabase : null; if (!cfg || !cfg.url || !cfg.anonKey || !lib || !lib.createClient) return null; try { PQPay.client = lib.createClient(cfg.url, cfg.anonKey, { auth: {persistSession: true, autoRefreshToken: true, detectSessionInUrl: true} }); PQPay.ready = true; return PQPay.client; } catch (err){ console.warn("[psycheiq] Supabase init failed:", err); return null; } } function pqFunctionsUrl(name){ const cfg = PQPay.cfg; if (!cfg) return ""; const base = cfg.functionsUrl || ((cfg.url || "").replace(/\/$/, "") + "/functions/v1"); return base.replace(/\/$/, "") + "/" + name; } /* Authorization header: the user's JWT when signed in, otherwise the publishable anon key so the function gateway still accepts the call. */ function pqAuthHeaders(){ const h = {"Content-Type": "application/json"}; const cfg = PQPay.cfg; if (cfg && cfg.anonKey) h.apikey = cfg.anonKey; const token = PQPay.session && PQPay.session.access_token; h.Authorization = "Bearer " + (token || (cfg && cfg.anonKey) || ""); return h; } async function pqCall(name, payload){ const url = pqFunctionsUrl(name); if (!url) throw new Error("Payments are not configured for this build."); const res = await fetch(url, { method: "POST", headers: pqAuthHeaders(), body: JSON.stringify(payload || {}) }); let data = null; try { data = await res.json(); } catch (_){ data = null; } if (!res.ok) throw new Error((data && data.error) || ("Request failed (" + res.status + ")")); return data || {}; } /* ---------------- session ---------------- */ async function pqRefreshSession(){ const c = pqInitSupabase(); if (!c) return null; const {data} = await c.auth.getSession(); PQPay.session = (data && data.session) || null; PQPay.user = PQPay.session ? PQPay.session.user : null; return PQPay.session; } function pqSignedIn(){ return Boolean(PQPay.user); } function pqEmail(){ return (PQPay.user && PQPay.user.email) || ""; } /* Magic-link sign in: no passwords to store, no reset flow to build. */ async function pqSendMagicLink(email){ const c = pqInitSupabase(); if (!c) throw new Error("Accounts are not configured for this build."); const {error} = await c.auth.signInWithOtp({ email: String(email || "").trim().toLowerCase(), options: {emailRedirectTo: location.origin + location.pathname} }); if (error) throw new Error(error.message); return true; } async function pqSignOut(){ const c = pqInitSupabase(); if (c) await c.auth.signOut(); PQPay.session = null; PQPay.user = null; State.pro = false; State.unlocked.clear(); State.lifetime = false; document.body.classList.remove("is-pro"); rerender(); toast("Signed out.
